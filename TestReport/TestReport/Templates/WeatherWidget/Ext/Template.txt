var WEATHER = /*WEATHER_JSON*/null;

var DAYS = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];
var DAYS_SHORT = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
var MONTHS = ["January", "February", "March", "April", "May", "June", "July",
              "August", "September", "October", "November", "December"];

function pad(n) { return (n < 10 ? "0" : "") + n; }

// ---------- Icons (inline SVG, 64x64) ----------
function sunSvg(cx, cy, r) {
  var rays = "";
  for (var i = 0; i < 8; i++) {
    var a = i * Math.PI / 4;
    rays += '<line x1="' + (cx + Math.cos(a) * (r + 5)) + '" y1="' + (cy + Math.sin(a) * (r + 5)) +
            '" x2="' + (cx + Math.cos(a) * (r + 11)) + '" y2="' + (cy + Math.sin(a) * (r + 11)) +
            '" stroke="#ffd25a" stroke-width="3" stroke-linecap="round"/>';
  }
  return '<g class="sun-rays">' + rays + '</g><circle cx="' + cx + '" cy="' + cy + '" r="' + r + '" fill="#ffd25a"/>';
}
function moonSvg() {
  return '<path d="M40 12 A20 20 0 1 0 52 44 A16 16 0 1 1 40 12 Z" fill="#f4f1c9"/>';
}
function cloudSvg(fill, dy) {
  dy = dy || 0;
  return '<g class="cloud-move"><path transform="translate(0,' + dy + ')" d="M18 48 h30 a10 10 0 0 0 0 -20 a14 14 0 0 0 -26 -4 a10 10 0 0 0 -4 24 z" fill="' + fill + '"/></g>';
}
function drops(color, symbol) {
  var s = "";
  for (var i = 0; i < 3; i++) {
    var x = 22 + i * 10;
    s += symbol === "snow"
      ? '<circle cx="' + x + '" cy="' + (55 + (i % 2) * 4) + '" r="2.4" fill="' + color + '"/>'
      : '<line x1="' + x + '" y1="52" x2="' + (x - 3) + '" y2="60" stroke="' + color + '" stroke-width="2.5" stroke-linecap="round"/>';
  }
  return s;
}
function weatherIcon(code, isDay) {
  var body;
  if (code === 0) {
    body = isDay ? sunSvg(32, 32, 12) : moonSvg();
  } else if (code <= 2) {
    body = (isDay ? sunSvg(24, 22, 9) : '<g transform="translate(-8,-6) scale(.8)">' + moonSvg() + '</g>') + cloudSvg("#ffffff", 4);
  } else if (code === 3) {
    body = cloudSvg("#dfe7ef", -4) + cloudSvg("#ffffff", 4);
  } else if (code <= 48) {
    body = cloudSvg("#e8edf2", -6) +
      '<line x1="12" y1="52" x2="52" y2="52" stroke="#e8edf2" stroke-width="3" stroke-linecap="round"/>' +
      '<line x1="18" y1="59" x2="46" y2="59" stroke="#e8edf2" stroke-width="3" stroke-linecap="round"/>';
  } else if (code <= 67 || (code >= 80 && code <= 82)) {
    body = cloudSvg("#e6edf5", -6) + drops("#7cc4ff", "rain");
  } else if (code <= 77 || code === 85 || code === 86) {
    body = cloudSvg("#ffffff", -6) + drops("#ffffff", "snow");
  } else {
    body = cloudSvg("#c9d3df", -6) + '<path d="M34 46 l-8 10 h7 l-4 9 l11 -13 h-7 l4 -6 z" fill="#ffd25a"/>';
  }
  return '<svg viewBox="0 0 64 64" width="100%" height="100%">' + body + '</svg>';
}
function weatherText(code) {
  if (code === 0) return "clear sky";
  if (code === 1) return "mainly clear";
  if (code === 2) return "partly cloudy";
  if (code === 3) return "overcast";
  if (code <= 48) return "fog";
  if (code <= 57) return "drizzle";
  if (code <= 67) return "rain";
  if (code <= 77) return "snow";
  if (code <= 82) return "rain showers";
  if (code <= 86) return "snow showers";
  return "thunderstorm";
}

// ---------- Clock ----------
(function buildTicks() {
  var g = document.getElementById("ticks"), s = "";
  for (var i = 0; i < 12; i++) {
    var a = i * Math.PI / 6, r1 = i % 3 === 0 ? 37 : 40;
    s += '<line x1="' + (50 + Math.sin(a) * r1) + '" y1="' + (50 - Math.cos(a) * r1) +
         '" x2="' + (50 + Math.sin(a) * 44) + '" y2="' + (50 - Math.cos(a) * 44) +
         '" stroke="rgba(255,255,255,0.8)" stroke-width="' + (i % 3 === 0 ? 2.5 : 1.2) + '" stroke-linecap="round"/>';
  }
  var wrap = document.createElement("div");
  wrap.innerHTML = '<svg xmlns="http://www.w3.org/2000/svg">' + s + '</svg>';
  var lines = wrap.firstChild.childNodes;
  while (lines.length) { g.appendChild(lines[0]); }
})();

function setHand(id, deg) {
  document.getElementById(id).setAttribute("transform", "rotate(" + deg + " 50 50)");
}

function dayPhase(h) {
  if (h >= 5 && h < 10) return "morning";
  if (h >= 10 && h < 17) return "day";
  if (h >= 17 && h < 21) return "evening";
  return "night";
}

function tick() {
  var d = new Date(), h = d.getHours(), m = d.getMinutes(), s = d.getSeconds();
  document.getElementById("hm").innerHTML = pad(h) + ":" + pad(m);
  document.getElementById("ss").innerHTML = pad(s);
  document.getElementById("date").innerHTML =
    DAYS[d.getDay()] + ", " + d.getDate() + " " + MONTHS[d.getMonth()] + " " + d.getFullYear();
  setHand("hh", (h % 12) * 30 + m * 0.5);
  setHand("mh", m * 6 + s * 0.1);
  setHand("sh", s * 6);
  var card = document.getElementById("card"), phase = dayPhase(h);
  if (card.className !== phase) card.className = phase;
}

var AVATAR_COLORS = ["#5b8def", "#f5a623", "#33b38a", "#e8607a", "#8e6cf1", "#20a4c4"];
var sortKey = "num", sortAsc = true;

function esc(t) {
  return String(t == null ? "" : t).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}
function initials(name) {
  var words = String(name).replace(/["'«»]/g, "").split(/\s+/), r = "";
  for (var i = 0; i < words.length && r.length < 2; i++) {
    if (words[i] && /[A-Za-zА-Яа-я0-9]/.test(words[i].charAt(0))) r += words[i].charAt(0).toUpperCase();
  }
  return r || "?";
}
function colorFor(name) {
  var h = 0, s = String(name);
  for (var i = 0; i < s.length; i++) { h = (h * 31 + s.charCodeAt(i)) % 997; }
  return AVATAR_COLORS[h % AVATAR_COLORS.length];
}
function ageText(iso) {
  if (!iso) return "";
  var p = iso.split("-"), d = new Date(+p[0], +p[1] - 1, +p[2]);
  var days = Math.floor((new Date() - d) / 86400000);
  if (days < 0) return "in " + (-days) + " d";
  if (days < 31) return days + " d ago";
  if (days < 365) return Math.floor(days / 30.4) + " mo ago";
  return Math.floor(days / 365) + " y ago";
}
function sortBy(key) {
  if (sortKey === key) { sortAsc = !sortAsc; } else { sortKey = key; sortAsc = true; }
  drawRows();
}
function drawRows() {
  var q = (document.getElementById("search").value || "").toLowerCase();
  var rows = [], i;
  for (i = 0; i < REPORT.rows.length; i++) {
    var r = REPORT.rows[i];
    if (!q || (r.client + " " + r.contract + " " + r.date).toLowerCase().indexOf(q) >= 0) rows.push(r);
  }
  rows.sort(function (a, b) {
    var x = sortKey === "date" ? a.iso : a[sortKey], y = sortKey === "date" ? b.iso : b[sortKey];
    if (typeof x === "string") { x = x.toLowerCase(); y = String(y).toLowerCase(); }
    return (x < y ? -1 : x > y ? 1 : 0) * (sortAsc ? 1 : -1);
  });
  var html = "";
  for (i = 0; i < rows.length; i++) {
    var row = rows[i];
    html += '<tr><td class="num">' + row.num + '</td>' +
      '<td><div class="client"><span class="avatar" style="background:' + colorFor(row.client) + '">' +
      esc(initials(row.client)) + '</span>' + esc(row.client) + '</div></td>' +
      '<td><span class="pill">' + esc(row.contract) + '</span></td>' +
      '<td class="cdate">' + esc(row.date) + '<span class="age">' + ageText(row.iso) + '</span></td></tr>';
  }
  if (!rows.length) html = '<tr><td colspan="4" class="empty">Nothing found</td></tr>';
  document.getElementById("rows").innerHTML = html;
  document.getElementById("count").innerHTML = rows.length + " of " + REPORT.rows.length + " contracts";
  var ths = document.getElementById("gridhead").getElementsByTagName("th");
  for (i = 0; i < ths.length; i++) {
    var k = ths[i].getAttribute("data-key");
    ths[i].className = k === sortKey ? "sorted" : "";
    ths[i].getElementsByTagName("span")[0].innerHTML = k === sortKey ? (sortAsc ? "&#9650;" : "&#9660;") : "&#8597;";
  }
}
